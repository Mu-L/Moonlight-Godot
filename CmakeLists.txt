cmake_minimum_required(VERSION 3.21)

# win平台默认使用静态链接库
if(CMAKE_SYSTEM_NAME STREQUAL "Windows" AND NOT DEFINED VCPKG_TARGET_TRIPLET)
    set(VCPKG_TARGET_TRIPLET "x64-windows-static-md" CACHE STRING "Target triplet")
endif()

# Android NDK bionic 缺乏 pthread_cancel，如果被部分 Unix 代码宏命中会导致编译失败，强行填补或忽略
if(CMAKE_SYSTEM_NAME STREQUAL "Android" OR ANDROID)
    add_compile_definitions("pthread_cancel(x)=0")
    # 强制防除 Windows 相关泄漏
    remove_definitions(-D_WIN32)
endif()

# 项目设置
set(LIBNAME "moonlight-godot" CACHE STRING "The name of the library")
set(GODOT_PROJECT_DIR "demo" CACHE STRING "The directory of a Godot project folder")

# vcpkg 工具链集成
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    if(DEFINED ENV{VCPKG_ROOT})
        set(CMAKE_TOOLCHAIN_FILE "$ENV{VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake" CACHE STRING "")
    endif()
endif()

# set(CMAKE_MAKE_PROGRAM ninja)

project(${LIBNAME} LANGUAGES CXX)

# 修复 Linux (特别是 ARM64) 下链接静态库 (如 ffmpeg) 到共享库时的重定位错误 (relocation R_AARCH64_ADR_PREL_PG_HI21)
# 强制所有目标（包括 vcpkg 构建的依赖项）使用位置无关代码 (PIC)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# 2. 平台与架构识别
if(ANDROID)
    set(TARGET_PLATFORM "android")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Windows")
    set(TARGET_PLATFORM "windows")
elseif(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set(TARGET_PLATFORM "ios")
elseif(APPLE)
    set(TARGET_PLATFORM "macos")
else()
    set(TARGET_PLATFORM "linux")
endif()

# 添加 -fPIC 编译选项
if(TARGET_PLATFORM STREQUAL "linux")
    set(CMAKE_SHARED_LINKER_FLAGS "-Wl,-Bsymbolic")
endif()

# 统一处理器架构名称
if(ANDROID)
    if(ANDROID_ABI STREQUAL "arm64-v8a")
        set(ACTUAL_ARCH "arm64")
    elseif(ANDROID_ABI STREQUAL "x86_64")
        set(ACTUAL_ARCH "x86_64")
    else()
        set(ACTUAL_ARCH "${CMAKE_SYSTEM_PROCESSOR}")
    endif()
else()
    # 修复 Windows 下 CMAKE_SYSTEM_PROCESSOR 为 AMD64 的情况
    if(CMAKE_SYSTEM_PROCESSOR STREQUAL "AMD64")
        set(ACTUAL_ARCH "x86_64")
    else()
        set(ACTUAL_ARCH "${CMAKE_SYSTEM_PROCESSOR}")
    endif()
endif()

if(ACTUAL_ARCH STREQUAL "aarch64")
    set(ACTUAL_ARCH "arm64")
endif()

set(PLATFORM_ID "${TARGET_PLATFORM}.${ACTUAL_ARCH}")
# 3. 查找依赖项

# --- FFmpeg ---
# 在Android上禁用FFmpeg的camera2ndk模块，否则会导致编译失败
set(FFMPEG_DEPENDENCY_camera2ndk_RELEASE "") 
find_package(FFMPEG REQUIRED)

# --- CURL ---
add_definitions(-DCURL_STATICLIB)
find_package(CURL REQUIRED)

# --- Godot-cpp ---
find_package(unofficial-godot-cpp CONFIG REQUIRED)

# --- Moonlight-common-c ---
find_package(moonlight-common-c CONFIG REQUIRED)

# 4. 库名前缀
if(DEFINED GODOTCPP_SUFFIX AND NOT GODOTCPP_SUFFIX STREQUAL "")
    set(GODOTCPP_SUFFIX_INTERNAL "${GODOTCPP_SUFFIX}")
elseif(DEFINED ENV{GODOTCPP_SUFFIX})
    set(GODOTCPP_SUFFIX_INTERNAL "$ENV{GODOTCPP_SUFFIX}")
else()
    get_target_property(PROP_SUFFIX unofficial::godot::cpp GODOTCPP_SUFFIX)
    if(PROP_SUFFIX AND NOT PROP_SUFFIX MATCHES "NOTFOUND")
        set(GODOTCPP_SUFFIX_INTERNAL "${PROP_SUFFIX}")
    else()
        # 后备方案：根据当前构建信息生成一个
        set(GODOTCPP_SUFFIX_INTERNAL ".${TARGET_PLATFORM}.template_debug.${ACTUAL_ARCH}")
    endif()
endif()

add_library(${LIBNAME} SHARED)

# 源文件
file(GLOB_RECURSE SOURCE_FILES CONFIGURE_DEPENDS "src/*.cpp" "src/*.h")
target_sources(${LIBNAME} PRIVATE ${SOURCE_FILES})

# ----------------------------------------------------------------------------
# Documentation Source Generation
# ----------------------------------------------------------------------------

find_package(Python3 REQUIRED COMPONENTS Interpreter)

set(DOC_CLASSES_DIR "${CMAKE_CURRENT_SOURCE_DIR}/doc_classes")
set(GEN_DOC_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/src/doc_classes.cpp")
set(DOC_GENERATOR_SCRIPT "${DOC_CLASSES_DIR}/doc_source_generator.py")

# 获取所有 XML 文件以便 CMake 跟踪依赖变化
file(GLOB DOC_XML_FILES "${DOC_CLASSES_DIR}/*.xml")

# 添加自定义命令生成源代码
add_custom_command(
    OUTPUT "${GEN_DOC_SOURCE}"
    COMMAND Python3::Interpreter "${DOC_GENERATOR_SCRIPT}" "${GEN_DOC_SOURCE}" "${DOC_CLASSES_DIR}"
    DEPENDS "${DOC_GENERATOR_SCRIPT}" ${DOC_XML_FILES}
    COMMENT "Generating doc_classes.cpp from XML documentation..."
    VERBATIM
)

# 确保生成的源文件被添加到目标中 (请将 your_target_name 替换为您实际的目标名称)
target_sources(${LIBNAME} PRIVATE "${GEN_DOC_SOURCE}")

# 编译特性与选项
target_compile_features(${LIBNAME} PRIVATE cxx_std_17)
target_include_directories(${LIBNAME} PRIVATE src)
target_include_directories(${LIBNAME} PRIVATE ${FFMPEG_INCLUDE_DIRS})

if(MSVC)
    set_property(TARGET ${LIBNAME} PROPERTY 
        MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

if(MSVC)
    target_compile_options(${LIBNAME} PRIVATE /W4)
    target_compile_definitions(${LIBNAME} PRIVATE CURL_STATICLIB)
else()
    target_compile_options(${LIBNAME} PRIVATE -Wall -Wextra)
endif()

# iOS 平台禁用 miniaudio 原生旁路，统一使用 Godot 音频系统。
if(TARGET_PLATFORM STREQUAL "ios")
    target_compile_definitions(${LIBNAME} PRIVATE NO_MINIAUDIO=1)
endif()

# 5. 链接库
target_link_libraries(${LIBNAME}
    PRIVATE 
    unofficial::godot::cpp
    CURL::libcurl
    moonlight-common-c::moonlight-common-c
    ${FFMPEG_LIBRARIES} 
    # 在 Linux/Android 上静态链接通常需要额外链接原子库或线程库
    $<$<PLATFORM_ID:Linux>:atomic m pthread dl>
    $<$<PLATFORM_ID:Android>:atomic m log>
)

# 6. 输出路径
if(WIN32)
    # Windows平台通常不需要lib前缀
    set_target_properties(${LIBNAME} PROPERTIES
        PREFIX ""
        OUTPUT_NAME "${LIBNAME}${GODOTCPP_SUFFIX_INTERNAL}"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin/${TARGET_PLATFORM}"
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin/${TARGET_PLATFORM}"
    )
else()
    # 非Windows平台添加lib前缀
    set_target_properties(${LIBNAME} PROPERTIES
        PREFIX "lib"
        OUTPUT_NAME "${LIBNAME}${GODOTCPP_SUFFIX_INTERNAL}"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin/${TARGET_PLATFORM}"
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin/${TARGET_PLATFORM}"
    )
endif()

# 7. 自动化部署
set(INSTALL_DEMO_DIR "${CMAKE_CURRENT_SOURCE_DIR}/${GODOT_PROJECT_DIR}/addons/${LIBNAME}/bin/${TARGET_PLATFORM}")
set(WORKSPACE_BIN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/bin/${TARGET_PLATFORM}")

# 部署主插件 DLL
add_custom_command(TARGET ${LIBNAME} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${INSTALL_DEMO_DIR}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different 
        "$<TARGET_FILE:${LIBNAME}>" 
        "${INSTALL_DEMO_DIR}/$<TARGET_FILE_NAME:${LIBNAME}>"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${WORKSPACE_BIN_DIR}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different 
        "$<TARGET_FILE:${LIBNAME}>" 
        "${WORKSPACE_BIN_DIR}/$<TARGET_FILE_NAME:${LIBNAME}>"
    COMMENT "Deploying plugin library to Godot project and workspace..."
)

# 部署运行时依赖 (仅在非静态构建时才需要)
if(WIN32)
    # 使用 file(GENERATE) 生成脚本以安全处理 TARGET_RUNTIME_DLLS 为空的情况
    set(DEPLOY_SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/deploy_dlls.cmake")
    file(GENERATE OUTPUT "${DEPLOY_SCRIPT}" CONTENT "
set(DLL_LIST \"$<TARGET_RUNTIME_DLLS:${LIBNAME}>\")
set(DEST_DIRS \"${INSTALL_DEMO_DIR}\" \"${WORKSPACE_BIN_DIR}\")
if(DLL_LIST)
    foreach(DEST \${DEST_DIRS})
        if(NOT EXISTS \"\${DEST}\")
            file(MAKE_DIRECTORY \"\${DEST}\")
        endif()
        foreach(DLL \${DLL_LIST})
            get_filename_component(FNAME \"\${DLL}\" NAME)
            execute_process(COMMAND \"${CMAKE_COMMAND}\" -E copy_if_different \"\${DLL}\" \"\${DEST}/\${FNAME}\")
        endforeach()
    endforeach()
endif()
")

    add_custom_command(TARGET ${LIBNAME} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -P "${DEPLOY_SCRIPT}"
        COMMENT "Deploying dependencies (if any)..."
    )
endif()
