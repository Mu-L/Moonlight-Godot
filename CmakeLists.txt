cmake_minimum_required(VERSION 3.21)

# win平台默认使用静态链接库
if(WIN32 AND NOT DEFINED VCPKG_TARGET_TRIPLET)
    set(VCPKG_TARGET_TRIPLET "x64-windows-static-md" CACHE STRING "Target triplet")
endif()

# 项目设置
set(LIBNAME "moonlight-godot" CACHE STRING "The name of the library")
set(GODOT_PROJECT_DIR "demo" CACHE STRING "The directory of a Godot project folder")

# vcpkg 工具链集成
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    if(DEFINED ENV{VCPKG_ROOT})
        set(CMAKE_TOOLCHAIN_FILE "$ENV{VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake" CACHE STRING "")
    endif()
endif()

project(${LIBNAME} LANGUAGES CXX)

# 修复 iOS 构建中 Ninja 生成器无法解析 'iphoneos' 的问题
# 必须在 project() 之后检查，确保覆盖 toolchain 的设置
if(CMAKE_SYSTEM_NAME STREQUAL "iOS" AND "${CMAKE_OSX_SYSROOT}" STREQUAL "iphoneos")
    execute_process(COMMAND xcrun --sdk iphoneos --show-sdk-path
        OUTPUT_VARIABLE IOS_SDK_PATH
        OUTPUT_STRIP_TRAILING_WHITESPACE
    )
    if(EXISTS "${IOS_SDK_PATH}")
        message(STATUS "Resolved 'iphoneos' sysroot to: ${IOS_SDK_PATH}")
        set(CMAKE_OSX_SYSROOT "${IOS_SDK_PATH}" CACHE PATH "Sysroot" FORCE)
        set(CMAKE_OSX_SYSROOT "${IOS_SDK_PATH}")
    endif()
endif()

# Make sure all the dependencies are satisfied
find_package(Python3 3.4 REQUIRED)
find_program(GIT git REQUIRED)

# Ensure godot-cpp submodule has been updated
if(NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/src/godot-cpp")
    message(NOTICE "godot-cpp bindings source not found")
    message(NOTICE "initializing/updating the godot-cpp submodule...")

    # update the c++ bindings submodule to populate it with the necessary source for the library
    execute_process(
        COMMAND git submodule update --init --recursive
        WORKING_DIRECTORY ${CMAKE_CURRENT_LIST_DIR}
        COMMAND_ERROR_IS_FATAL ANY
    )
endif()
add_subdirectory(src/godot-cpp SYSTEM)

# Add godot-cpp's module path and include the exported functions.
set(CMAKE_MODULE_PATH "${CMAKE_MODULE_PATH};${godot-cpp_SOURCE_DIR}/cmake")
include(GodotCPPModule)

# Retrieve properties from godot-cpp
get_target_property(GODOTCPP_SUFFIX godot::cpp GODOTCPP_SUFFIX)
get_target_property(GODOTCPP_PLATFORM godot::cpp GODOTCPP_PLATFORM)



# 3. 查找依赖项

# --- FFmpeg ---
# 在Android上禁用FFmpeg的camera2ndk模块，否则会导致编译失败
set(FFMPEG_DEPENDENCY_camera2ndk_RELEASE "") 
find_package(FFMPEG REQUIRED)

# --- CURL ---
add_definitions(-DCURL_STATICLIB)
find_package(CURL REQUIRED)

# --- Godot-cpp ---
# Replaced unofficial-godot-cpp with submodule approach above
# find_package(unofficial-godot-cpp CONFIG REQUIRED)

# --- Moonlight-common-c ---
find_package(moonlight-common-c CONFIG REQUIRED)

# 4. 库名前缀
if(DEFINED GODOTCPP_SUFFIX AND NOT GODOTCPP_SUFFIX STREQUAL "")
    set(GODOTCPP_SUFFIX_INTERNAL "${GODOTCPP_SUFFIX}")
elseif(DEFINED ENV{GODOTCPP_SUFFIX})
    set(GODOTCPP_SUFFIX_INTERNAL "$ENV{GODOTCPP_SUFFIX}")
# else()
#     get_target_property(PROP_SUFFIX unofficial::godot::cpp GODOTCPP_SUFFIX)
#     if(PROP_SUFFIX AND NOT PROP_SUFFIX MATCHES "NOTFOUND")
#         set(GODOTCPP_SUFFIX_INTERNAL "${PROP_SUFFIX}")
#     else()
#         # 后备方案：根据当前构建信息生成一个
#         set(GODOTCPP_SUFFIX_INTERNAL ".${TARGET_PLATFORM}.template_debug.${ACTUAL_ARCH}")
#     endif()
endif()

add_library(${LIBNAME} SHARED)

# 源文件
file(GLOB_RECURSE SOURCE_FILES CONFIGURE_DEPENDS "src/*.cpp" "src/*.h")
target_sources(${LIBNAME} PRIVATE ${SOURCE_FILES})

# Fetch a list of the xml files to use for documentation and add to our target
file(GLOB_RECURSE DOC_XML LIST_DIRECTORIES NO CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/doc_classes/*.xml")

# conditionally add doc data to compile output
if(DOC_XML)
    if(GODOTCPP_TARGET MATCHES "editor|template_debug")
        target_doc_sources(${LIBNAME} ${DOC_XML})
    endif()
endif()

# 编译特性与选项
target_compile_features(${LIBNAME} PRIVATE cxx_std_17)
target_include_directories(${LIBNAME} PRIVATE src)
target_include_directories(${LIBNAME} PRIVATE ${FFMPEG_INCLUDE_DIRS})

if(MSVC)
    set_property(TARGET ${LIBNAME} PROPERTY 
        MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
endif()

if(MSVC)
    target_compile_options(${LIBNAME} PRIVATE /W4 /wd4244)
    target_compile_definitions(${LIBNAME} PRIVATE CURL_STATICLIB)
else()
    target_compile_options(${LIBNAME} PRIVATE -Wall -Wextra)
endif()

# 5. 链接库
target_link_libraries(${LIBNAME}
    PRIVATE 
    godot::cpp
    CURL::libcurl
    moonlight-common-c::moonlight-common-c
    ${FFMPEG_LIBRARIES} 
    # 在 Linux/Android 上静态链接通常需要额外链接原子库或线程库
    $<$<STREQUAL:${GODOTCPP_PLATFORM},linux>:atomic m pthread dl>
    $<$<STREQUAL:${GODOTCPP_PLATFORM},android>:atomic m log>
)

# 6. 输出路径
set_target_properties(${LIBNAME} PROPERTIES
    PREFIX ""
    OUTPUT_NAME "${LIBNAME}${GODOTCPP_SUFFIX}"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin/${GODOTCPP_PLATFORM}"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin/${GODOTCPP_PLATFORM}"
)

# 7. 自动化部署
set(INSTALL_DEMO_DIR "${CMAKE_CURRENT_SOURCE_DIR}/${GODOT_PROJECT_DIR}/addons/${LIBNAME}/bin/${GODOTCPP_PLATFORM}")
set(WORKSPACE_BIN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/bin/${GODOTCPP_PLATFORM}")

# 部署主插件 DLL
add_custom_command(TARGET ${LIBNAME} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${INSTALL_DEMO_DIR}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different 
        "$<TARGET_FILE:${LIBNAME}>" 
        "${INSTALL_DEMO_DIR}/$<TARGET_FILE_NAME:${LIBNAME}>"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${WORKSPACE_BIN_DIR}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different 
        "$<TARGET_FILE:${LIBNAME}>" 
        "${WORKSPACE_BIN_DIR}/$<TARGET_FILE_NAME:${LIBNAME}>"
    COMMENT "Deploying plugin library to Godot project and workspace..."
)

# 部署运行时依赖 (仅在非静态构建时才需要)
if(WIN32)
    # 使用 file(GENERATE) 生成脚本以安全处理 TARGET_RUNTIME_DLLS 为空的情况
    set(DEPLOY_SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/deploy_dlls.cmake")
    file(GENERATE OUTPUT "${DEPLOY_SCRIPT}" CONTENT "
set(DLL_LIST \"$<TARGET_RUNTIME_DLLS:${LIBNAME}>\")
set(DEST_DIRS \"${INSTALL_DEMO_DIR}\" \"${WORKSPACE_BIN_DIR}\")
if(DLL_LIST)
    foreach(DEST \${DEST_DIRS})
        if(NOT EXISTS \"\${DEST}\")
            file(MAKE_DIRECTORY \"\${DEST}\")
        endif()
        foreach(DLL \${DLL_LIST})
            get_filename_component(FNAME \"\${DLL}\" NAME)
            execute_process(COMMAND \"${CMAKE_COMMAND}\" -E copy_if_different \"\${DLL}\" \"\${DEST}/\${FNAME}\")
        endforeach()
    endforeach()
endif()
")

    add_custom_command(TARGET ${LIBNAME} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -P "${DEPLOY_SCRIPT}"
        COMMENT "Deploying dependencies (if any)..."
    )
endif()
